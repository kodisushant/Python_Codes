def book_seat(st,shw,scr):
    os.chdir('C:\\Users\\susha\\Downloads\\my python codes\\theatre\\operation\\'+scr+"\\"+shw+"\\"+st)
    # print(os.getcwd())
    print("\n")
    key = 1
    count = 0
    while key:
        print("The available seats are:")
        with open("seats.txt","r") as p: 
            seats = p.read()
        print(seats)
        print("\n0=available , X=booked")

        
        row = int(input("Enter the row number:"))
        col = int(input("Enter the col number:"))
        if row <= 0 or row >5 or col<=0 or col>10: 
            key = 1 
            continue
        mat = []
        rows = seats.strip().split("\n")
        mat = [r.split() for r in rows]

        if mat[row-1][col-1] == "0":
            mat[row-1][col-1]="X"
            count+=1
        else: 
            print("sorry the seat is already booked")
        
        with open("seats.txt", "w") as p:
            for r in mat:
                p.write(" ".join(r) + "\n")
        key = int(input("Press (any number) to book more seats else Press 0:"))
    return count
